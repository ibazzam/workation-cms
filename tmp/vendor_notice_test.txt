            function locationSelectionPayload() {\n
                const country = String(locationCountry && locationCountry.value || '').trim();\n
                const state = String(locationState && locationState.value || '').trim();\n
                const city = String(locationCity && locationCity.value || '').trim();\n
                const queryParts = [city, state, country].filter(Boolean);\n
                const query = queryParts.join(', ');\n
                const hasCity = city !== '';\n
                const hasState = state !== '';\n
                return { country, state, city, query, hasCity, hasState };\n

            async function geocodeMapSelection(query) {\n
                const cacheKey = String(query || '').trim().toLowerCase();\n
                if (cacheKey === '') {\n
                    return null;\n
                }\n
                if (mapGeocodeCache.has(cacheKey)) {\n
                    return mapGeocodeCache.get(cacheKey);\n
                }\n
\n
                const url = 'https://nominatim.openstreetmap.org/search?format=jsonv2&limit=1&q=' + encodeURIComponent(query);\n
                try {\n
                    const response = await fetch(url, {\n
                        method: 'GET',\n
                        headers: {\n
                            Accept: 'application/json',\n
                        },\n
                        cache: 'force-cache',\n
                    });\n
                    if (!response.ok) {\n
                        return null;\n
                    }\n
                    const rows = await response.json();\n
                    if (!Array.isArray(rows) || rows.length === 0) {\n
                        return null;\n
                    }\n
                    const first = rows[0] || {};\n
                    const lat = Number(first.lat);\n
                    const lng = Number(first.lon);\n
                    if (!Number.isFinite(lat) || !Number.isFinite(lng)) {\n
                        return null;\n
                    }\n
                    const point = [lat, lng];\n
                    mapGeocodeCache.set(cacheKey, point);\n
                    return point;\n
                } catch (error) {\n
                    return null;\n
                }\n
            }\n

            async function centerMapForLocationSelection(forceLookup) {\n
                if (!locationMapContext || !locationMapContext.map) {\n
                    return;\n
                }\n
\n
                const map = locationMapContext.map;\n
                const payload = locationSelectionPayload();\n
                const fallback = fallbackMapView(payload.country);\n
                const fallbackPoint = [fallback[0], fallback[1]];\n
                const fallbackZoom = payload.hasCity ? 11 : (payload.hasState ? 9 : fallback[2]);\n
\n
                // Move immediately so the map always responds even if geocoding is slow.\n
                map.flyTo(fallbackPoint, fallbackZoom, { animate: true, duration: 0.28 });\n
\n
                if (payload.query === '') {\n
                    return;\n
                }\n
\n
                const requestId = ++mapLookupRequestId;\n
                let targetPoint = null;\n
                if (forceLookup || payload.hasCity || payload.hasState) {\n
                    targetPoint = await geocodeMapSelection(payload.query);\n
                }\n
                if (requestId !== mapLookupRequestId) {\n
                    return;\n
                }\n
\n
                if (!targetPoint) {\n
                    return;\n
                }\n
\n
                const zoom = payload.hasCity ? 13 : (payload.hasState ? 10 : fallback[2]);\n
                map.flyTo(targetPoint, zoom, { animate: true, duration: 0.45 });\n
            }\n

            async function run(path, triggerButton) {\n
                if (!output) {\n
                    return;\n
                }\n
                const token = getToken();\n
                if (!token) {\n
                    setState("warn", "TOKEN REQUIRED");\n
                    setMeta("Save a vendor token before running requests.");\n
                    output.textContent = "Save a vendor token first.";\n
                    return;\n
                }\n
\n
                const verdict = evaluateToken(token);\n
                if (!verdict.isUsable) {\n
                    setState(verdict.stateType, verdict.stateText);\n
                    setMeta(verdict.metaText);\n
                    output.textContent = "Token is expired or invalid. Save a fresh vendor token first.";\n
                    return;\n
                }\n
\n
                const button = triggerButton || null;\n
                if (button) {\n
                    button.disabled = true;\n
                    button.classList.add("is-loading");\n
                    if (!button.dataset.label) {\n
                        button.dataset.label = button.textContent || "Run";\n
                    }\n
                    button.textContent = "Running";\n
                }\n
\n
                output.textContent = "Loading " + path + " ...";\n
                try {\n
                    const res = await fetch(apiBase + path, {\n
                        method: "GET",\n
                        headers: {\n
                            "Authorization": "Bearer " + token,\n
                            "Accept": "application/json"\n
                        },\n
                        cache: "no-store"\n
                    });\n
                    const text = await res.text();\n
                    let parsed = text;\n
                    try {\n
                        parsed = JSON.stringify(JSON.parse(text), null, 2);\n
                    } catch (error) {\n
                        // Keep plain text if response is not JSON.\n
                    }\n
                    output.textContent = "Status: " + res.status + "\n\n" + parsed;\n
                    if (res.ok) {\n
                        applyTokenFeedback(token, "ok", "TOKEN VALID");\n
                    } else if (res.status === 401 || res.status === 403) {\n
                        setState("err", "TOKEN INVALID FOR VENDOR");\n
                        setMeta("The API rejected this token for vendor access.");\n
                    } else {\n
                        applyTokenFeedback(token, "warn", "REQUEST COMPLETED WITH WARNINGS");\n
                    }\n
                } catch (error) {\n
                    setState("err", "REQUEST FAILED");\n
                    setMeta("Request failed before token validation could complete.");\n
                    output.textContent = "Network/CORS error. Ensure API allows origin https://www.workation.mv\n\n" + String(error);\n
                } finally {\n
                    if (button) {\n
                        button.disabled = false;\n
                        button.classList.remove("is-loading");\n
                        button.textContent = button.dataset.label || "Run";\n
                    }\n
                }\n
            }\n

                function normalizeCalendarDayState(rawState) {\n
                    const inventory = Number(rawState && rawState.inventory ? rawState.inventory : 0);\n
                    const reserved = Number(rawState && rawState.reserved ? rawState.reserved : 0);\n
                    const isClosed = Boolean(rawState && rawState.is_closed);\n

                    const isFullyReserved = inventory > 0 && reserved >= inventory;\n

                        inventory,\n
                        reserved,\n
                        isClosed,\n
                        hasBooking,\n
                        isBooked: hasBooking || isFullyReserved,\n

                async function submitAvailabilityCalendarToggle(form, dateKey, shouldClose, currentState) {\n
                    if (!form || !dateKey) {\n
                        return;\n
                    }\n
\n
                    const listingCategoryInput = form.querySelector('input[name="listing_category"]');\n
                    const listingCategory = String(listingCategoryInput ? listingCategoryInput.value : '').trim().toLowerCase();\n
                    const roomInput = form.querySelector('[data-availability-role="room"]');\n
                    if (listingCategory === 'accommodation' && String(roomInput ? roomInput.value : '').trim() === '') {\n
                        window.alert('Select a room first before editing accommodation dates.');\n
                        return;\n
                    }\n
\n
                    const calendarMeta = availabilityCalendarByForm.get(form);\n
                    if (!calendarMeta || calendarMeta.isSaving) {\n
                        return;\n
                    }\n
                    calendarMeta.isSaving = true;\n
\n
                    const inventoryInput = form.querySelector('input[name="inventory"]');\n
                    const parsedInventory = Number(currentState && Number.isFinite(currentState.inventory)\n
                        ? currentState.inventory\n
                        : Number(inventoryInput ? inventoryInput.value : 0));\n
                    const nextInventory = shouldClose\n
                        ? Math.max(0, Number.isFinite(parsedInventory) ? parsedInventory : 0)\n
                        : Math.max(1, Number.isFinite(parsedInventory) ? parsedInventory : 1);\n
\n
                    const dateInput = form.querySelector('input[name="slot_date"]');\n
                    const rangeFromInput = form.querySelector('input[name="apply_range_from"]');\n
                    const rangeToInput = form.querySelector('input[name="apply_range_to"]');\n
                    const scheduleProfileInput = form.querySelector('select[name="schedule_profile"]');\n
                    const closedSelect = form.querySelector('select[name="is_closed"]');\n
                    if (inventoryInput) inventoryInput.value = String(nextInventory);\n
                    if (dateInput) dateInput.value = dateKey;\n
                    if (rangeFromInput) rangeFromInput.value = dateKey;\n
                    if (rangeToInput) rangeToInput.value = dateKey;\n
                    if (scheduleProfileInput) scheduleProfileInput.value = 'one_off';\n
                    if (closedSelect) closedSelect.value = shouldClose ? '1' : '0';\n
\n
                    const formData = new FormData(form);\n
                    formData.set('slot_date', dateKey);\n
                    formData.set('apply_range_from', dateKey);\n
                    formData.set('apply_range_to', dateKey);\n
                    formData.set('schedule_profile', 'one_off');\n
                    formData.set('inventory', String(nextInventory));\n
                    formData.set('is_closed', shouldClose ? '1' : '0');\n
\n
                    const csrfToken = String(formData.get('_token') || '');\n
                    try {\n
                        const response = await fetch(String(form.getAttribute('action') || '/portal/vendor/availability/save'), {\n
                            method: 'POST',\n
                            credentials: 'same-origin',\n
                            headers: {\n
                                'X-Requested-With': 'XMLHttpRequest',\n
                                'X-CSRF-TOKEN': csrfToken,\n
                            },\n
                            body: formData,\n
                        });\n
\n
                        if (!response.ok) {\n
                            throw new Error('Update failed with status ' + response.status);\n
                        }\n
\n
                        const targetKey = selectedAvailabilityTargetKey(form);\n
                        if (!calendarMeta.states[targetKey] || typeof calendarMeta.states[targetKey] !== 'object') {\n
                            calendarMeta.states[targetKey] = {};\n
                        }\n
                        const existingRaw = calendarMeta.states[targetKey][dateKey] || {};\n
                        calendarMeta.states[targetKey][dateKey] = {\n
                            inventory: nextInventory,\n
                            reserved: Number(existingRaw.reserved || 0),\n
                            is_closed: shouldClose,\n
                            has_booking: Boolean(existingRaw.has_booking),\n
                        };\n
                    } catch (error) {\n
                        console.error('Unable to save availability update', error);\n
                        window.alert('Could not save this availability change. Please try again.');\n
                    } finally {\n
                        calendarMeta.isSaving = false;\n
                        renderAvailabilityCalendar(form);\n
                    }\n
                }\n
